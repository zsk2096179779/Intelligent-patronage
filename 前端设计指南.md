# 组合产品创建 - 前端设计指南

## 概述

本文档说明前端如何配合后端实现组合产品的完整创建流程，包括多步骤配置、草稿保存、提交审核等功能。

## 整体流程设计

### 流程图

```
┌─────────────────┐
│  1. 选择策略     │ → GET /api/strategies (下拉框选择)
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  2. 创建组合     │ → POST /api/strategy-combination
│  (基础信息)      │   返回 portfolioId
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  3. 配置基础信息 │ → PUT /api/strategy-combination/{id}/basic-info
│  (可选，可修改)  │   保存草稿
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  4. 配置产品参数 │ → POST /api/strategy-combination/{id}/product-params
│  (费率、金额等)  │   保存草稿
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  5. 配置基金持仓 │ → POST /api/strategy-combination/{id}/holdings
│  (选择基金+权重) │   验证权重总和=100%
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  6. 预览确认     │ → GET /api/strategy-combination/{id}/product-params
│                 │   GET /api/strategy-combination/{id}/holdings
└────────┬────────┘
         │
         ▼
┌─────────────────┐
│  7. 提交审核     │ → POST /api/strategy-combination/{id}/submit
│  (status变更为   │   status: draft → pending_review
│   pending_review)│
└─────────────────┘
```

## 详细步骤设计

### 步骤1：选择策略（创建组合前）

**页面元素：**
- 策略下拉框（必填）
- 组合名称输入框（必填）
- 风险等级选择（可选）
- 策略类型选择（可选）

**接口调用：**
```javascript
// 1. 加载策略列表
GET /api/strategies
// 返回：{ code: 200, data: [{ strategyId, strategyName, strategyType, strategyRefId, ... }] }

// 2. 用户选择策略后，创建组合
POST /api/strategy-combination
// 请求体：
{
  "name": "稳健理财组合",
  "riskLevel": "中低风险",
  "strategyType": "大类资产配置",
  "strategyRefId": 1001,  // 从策略列表中选择的 strategyRefId
  "summary": "组合简介（可选）",
  "targetInvestor": "目标客户（可选）"
}
// 返回：{ code: 200, data: { portfolioId: 1 } }
```

**前端逻辑：**
```javascript
// 伪代码示例
async function step1() {
  // 1. 加载策略列表
  const strategies = await fetch('/api/strategies');
  // 填充下拉框
  
  // 2. 用户填写表单后，创建组合
  const formData = {
    name: form.name.value,
    riskLevel: form.riskLevel.value,
    strategyType: form.strategyType.value,
    strategyRefId: form.strategyRefId.value, // 从下拉框选择
    summary: form.summary.value,
    targetInvestor: form.targetInvestor.value
  };
  
  const result = await fetch('/api/strategy-combination', {
    method: 'POST',
    body: JSON.stringify(formData)
  });
  
  if (result.code === 200) {
    const portfolioId = result.data.portfolioId;
    // 保存 portfolioId 到状态管理（Vuex/Redux）或 localStorage
    // 跳转到下一步
    router.push(`/portfolio/${portfolioId}/step2`);
  }
}
```

### 步骤4：配置基金持仓（关键步骤）

**页面元素：**
- 基金选择器（支持搜索、多选）
- 持仓表格：
  - 基金代码
  - 基金名称
  - 权重（%）
  - 备注
  - 操作（删除）
- 权重总和显示：`总权重：XX%`（实时计算）
- 添加基金按钮

**接口调用：**
```javascript
// 保存持仓
POST /api/strategy-combination/{portfolioId}/holdings
// 请求体：数组
[
  {
    "fundCode": "000001",
    "fundName": "基金名称1",
    "weight": 30.00,
    "remark": "备注1"
  },
  {
    "fundCode": "000002",
    "fundName": "基金名称2",
    "weight": 70.00,
    "remark": "备注2"
  }
]
// 注意：权重总和必须为100%（或 1.0），否则返回400错误；后台会自动兼容 0-100 与 0-1 的传入格式

// 查询持仓（回显用）
GET /api/strategy-combination/{portfolioId}/holdings
// 返回：{ code: 200, data: [{ id, fundCode, fundName, weight, remark }] }
```

**前端逻辑：**
```javascript
// 伪代码
let holdings = []; // 持仓列表

// 1. 加载已有持仓
async function loadHoldings(portfolioId) {
  const result = await fetch(`/api/strategy-combination/${portfolioId}/holdings`);
  if (result.code === 200) {
    holdings = result.data || [];
    renderHoldingsTable();
  }
}

// 2. 添加基金
function addFund(fundCode, fundName) {
  holdings.push({
    fundCode: fundCode,
    fundName: fundName,
    weight: 0,
    remark: ''
  });
  renderHoldingsTable();
  updateWeightSum();
}

// 3. 更新权重（实时验证）
function updateWeight(index, newWeight) {
  holdings[index].weight = parseFloat(newWeight);
  updateWeightSum();
  
  // 实时显示权重总和
  const sum = holdings.reduce((total, h) => total + (h.weight || 0), 0);
  document.getElementById('weightSum').textContent = `总权重：${sum.toFixed(2)}%`;
  
  // 如果总和接近100%（或 1.0），显示提示
  if (Math.abs(sum - 100) < 0.01) {
    document.getElementById('weightSum').classList.add('success');
  } else {
    document.getElementById('weightSum').classList.remove('success');
  }
}

// 4. 保存持仓
async function saveHoldings(portfolioId) {
  // 验证权重总和
  const sum = holdings.reduce((total, h) => total + (h.weight || 0), 0);
  if (Math.abs(sum - 100) > 0.01) {
    alert('持仓权重总和必须为100%（或 1.0），当前为：' + sum.toFixed(2) + '%');
    return false;
  }
  
  const result = await fetch(`/api/strategy-combination/${portfolioId}/holdings`, {
    method: 'POST',
    body: JSON.stringify(holdings)
  });
  
  if (result.code === 200) {
    return true;
  } else {
    alert(result.message);
    return false;
  }
}
```

**权重验证提示：**
```javascript
// 实时显示权重总和和状态（示例仍使用百分比展示，可按需改为 0-1 小数）
function updateWeightSum() {
  const sum = holdings.reduce((total, h) => total + (h.weight || 0), 0);
  const sumElement = document.getElementById('weightSum');
  
  sumElement.textContent = `总权重：${sum.toFixed(2)}%`;
  
  if (Math.abs(sum - 100) < 0.01) {
    sumElement.className = 'weight-sum success'; // 绿色
    sumElement.textContent += ' ✓';
  } else if (sum > 100) {
    sumElement.className = 'weight-sum error'; // 红色
    sumElement.textContent += ' (超出)';
  } else {
    sumElement.className = 'weight-sum warning'; // 黄色
    sumElement.textContent += ' (不足)';
  }
}
```

## 状态管理建议

### 使用状态管理库（Vuex/Redux/Pinia）

```javascript
// 示例：Vuex Store
const portfolioStore = {
  state: {
    currentPortfolioId: null,
    currentStep: 1,
    basicInfo: {},
    productParams: {},
    holdings: [],
    strategies: [] // 策略列表
  },
  
  mutations: {
    SET_PORTFOLIO_ID(state, id) {
      state.currentPortfolioId = id;
    },
    SET_CURRENT_STEP(state, step) {
      state.currentStep = step;
    },
    SET_BASIC_INFO(state, info) {
      state.basicInfo = info;
    },
    SET_PRODUCT_PARAMS(state, params) {
      state.productParams = params;
    },
    SET_HOLDINGS(state, holdings) {
      state.holdings = holdings;
    },
    SET_STRATEGIES(state, strategies) {
      state.strategies = strategies;
    }
  },
  
  actions: {
    async createPortfolio({ commit }, formData) {
      const result = await api.createPortfolio(formData);
      if (result.code === 200) {
        commit('SET_PORTFOLIO_ID', result.data.portfolioId);
        commit('SET_BASIC_INFO', formData);
        return result.data.portfolioId;
      }
      throw new Error(result.message);
    },
    
    async saveBasicInfo({ commit, state }, formData) {
      const result = await api.updateBasicInfo(state.currentPortfolioId, formData);
      if (result.code === 200) {
        commit('SET_BASIC_INFO', formData);
      }
      return result;
    },
    
    // ... 其他 actions
  }
};
```

### 或使用 localStorage（简单场景）

```javascript
// 保存草稿到本地
function saveDraft(portfolioId, step, data) {
  const key = `portfolio_draft_${portfolioId}`;
  const draft = JSON.parse(localStorage.getItem(key) || '{}');
  draft[step] = data;
  draft.lastUpdate = Date.now();
  localStorage.setItem(key, JSON.stringify(draft));
}

// 加载草稿
function loadDraft(portfolioId) {
  const key = `portfolio_draft_${portfolioId}`;
  return JSON.parse(localStorage.getItem(key) || '{}');
}
```

```javascript
// 实时验证
function validateForm(formData, step) {
  const errors = {};
  
  if (step === 1) {
    if (!formData.name || formData.name.trim() === '') {
      errors.name = '组合名称不能为空';

    }
  }
  
  if (step === 4) {

    if (Math.abs(sum - 100) > 0.01) {
      errors.holdings = '持仓权重总和必须为100%（或 1.0）';
```

```javascript
async function apiCall(url, options) {
  try {
    const response = await fetch(url, options);
    const result = await response.json();
    
    if (result.code === 200) {
      return result;
    } else if (result.code === 400) {
      // 参数错误，显示提示

  }
}
```

```javascript
        strategyType: this.formData.strategyType,
        summary: this.formData.summary,
        targetInvestor: this.formData.targetInvestor
      });
      
      if (result.code === 200) {
        this.portfolioId = result.data.portfolioId;
        this.currentStep = 2;
        this.$router.push(`/portfolio/create/step2/${this.portfolioId}`);
      }
    },
    
    // 步骤2：保存基础信息
    async handleStep2() {
      const result = await api.updateBasicInfo(this.portfolioId, {
        name: this.formData.name,
        riskLevel: this.formData.riskLevel,
        strategyType: this.formData.strategyType,
        summary: this.formData.summary,
        targetInvestor: this.formData.targetInvestor,
        strategyId: this.formData.strategyRefId
      });
      
      if (result.code === 200) {
        this.currentStep = 3;
        this.$router.push(`/portfolio/create/step3/${this.portfolioId}`);
      }
    },
    
    // 步骤3：保存产品参数
    async handleStep3() {
      const result = await api.saveProductParams(this.portfolioId, this.formData.productParams);
      
      if (result.code === 200) {
        this.currentStep = 4;
        this.$router.push(`/portfolio/create/step4/${this.portfolioId}`);
      }
    },
    
    // 步骤4：保存持仓
    async handleStep4() {
      // 验证权重
      const sum = this.formData.holdings.reduce((t, h) => t + (h.weight || 0), 0);
      if (Math.abs(sum - 100) > 0.01) {
        this.$message.error('持仓权重总和必须为100%（或 1.0）');
        return;
      }
      
      const result = await api.saveHoldings(this.portfolioId, this.formData.holdings);
      
      if (result.code === 200) {
        this.currentStep = 5;
        this.$router.push(`/portfolio/create/preview/${this.portfolioId}`);
      }
    },
    
    // 步骤5：提交审核
    async handleSubmit() {
      const result = await api.submitForReview(this.portfolioId);
      
      if (result.code === 200) {
        this.$message.success('提交审核成功');
        this.$router.push('/portfolio/list');
      }
    },
    
    // 保存草稿（不跳转）
    async saveDraft() {
      // 根据当前步骤保存对应数据
      if (this.currentStep >= 2) {
        await api.updateBasicInfo(this.portfolioId, {...});
      }
      if (this.currentStep >= 3) {
        await api.saveProductParams(this.portfolioId, this.formData.productParams);
      }
      if (this.currentStep >= 4) {
        await api.saveHoldings(this.portfolioId, this.formData.holdings);
      }
      this.$message.success('草稿已保存');
    }
  }
};
```

## 关键注意事项

### 1. 权重验证
- **实时验证**：用户输入权重时实时计算总和
- **提交前验证**：保存前再次验证，确保总和为100%
- **容差处理**：允许0.01%的误差（浮点数精度问题）

### 2. 状态管理
- **组合ID保存**：创建组合后立即保存 `portfolioId`，后续步骤都需要
- **步骤状态**：记录当前步骤，支持前进/后退
- **数据回显**：编辑草稿时，需要加载已保存的数据

### 3. 错误处理
- **网络错误**：显示友好提示，允许重试
- **参数错误**：显示具体错误信息（如权重总和不为100%）
- **状态错误**：提交审核时，如果状态不是 `draft`，显示提示

### 4. 用户体验
- **自动保存**：定期自动保存草稿，避免数据丢失
- **进度提示**：显示当前步骤和完成进度
- **数据验证**：实时验证，及时提示错误
- **加载状态**：数据加载时显示加载动画
